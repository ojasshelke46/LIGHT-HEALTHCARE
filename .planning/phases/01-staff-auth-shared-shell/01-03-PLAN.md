   ```typescript
   import { redirect } from "next/navigation";
   import { createClient } from "@/lib/supabase/server";
   import type { StaffRole } from "@light/shared-types";
   export type StaffIdentity = { id: string; name: string; role: StaffRole };
   export async function getStaff(): Promise<StaffIdentity> {
     const supabase = await createClient();
     const { data: { user } } = await supabase.auth.getUser();
     if (!user) redirect("/login");
     const { data } = await supabase.from("staff").select("id, name, role").eq("auth_user_id", user.id).single();
     if (!data) redirect("/login");
     return { id: data.id, name: data.name, role: data.role };
   }
   ```

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| browser cookie → server layout | The signed-in session cookie is trusted only after server-side `getUser()` revalidation |
| requested URL → role segment | A staff member may request a portal that is not theirs |
| client component ← server layout | `staff` identity is passed from a server component into a client DashboardLayout |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-01-08 | Elevation of privilege | cross-role portal access | mitigate | Enforced by existing `src/middleware.ts` (KEPT); the role `layout.tsx` additionally derives role from the server `staff` read, not from the URL or client — defense in depth |
| T-01-09 | Spoofing | session identity in layout | mitigate | `getStaff()` uses `supabase.auth.getUser()` (server revalidation), not `getSession()`; unauthenticated or non-staff users are redirected to /login |
| T-01-10 | Information disclosure | staff identity passed to client | accept | Only `{ id, name, role }` is passed to the client shell (name is shown by design); no email/phone/PII beyond the displayed name crosses to the client |
| T-01-11 | Denial of service | per-request staff read | accept | One indexed `staff` lookup per navigation (by `auth_user_id`); RLS restricts the row to the owner; acceptable for internal staff volume |
| T-01-12 | Repudiation | logout | accept | `signOut()` clears the session server-side via Supabase; audit trail deferred (v2) |
</threat_model>
